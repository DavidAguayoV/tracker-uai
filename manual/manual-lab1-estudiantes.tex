%%=============================================================================
\section{¿Qué es y para qué sirve?}
%%=============================================================================

\textbf{Tracker UAI} es un programa que se abre en tu navegador y sirve para
\textbf{analizar videos de experimentos}. En el Laboratorio~1 grabarás el
movimiento con tu celular y el programa \textbf{seguirá el objeto cuadro a
cuadro} para entregarte su \textbf{posición, velocidad y aceleración}, además de
gráficos y el \textbf{ajuste de un modelo} (por ejemplo, la aceleración $a$ con
su incertidumbre). Todo se puede \textbf{exportar} para el informe.

%%=============================================================================
\section{Instalación (una sola vez)}
%%=============================================================================

\begin{enumerate}
  \item \Accion Instala \textbf{Python 3.11 o superior} desde
    \texttt{python.org/downloads}. En Windows, \textbf{marca «Add Python to
    PATH»} durante la instalación.
  \item \Accion Descomprime la carpeta del programa en un lugar fácil (Escritorio
    o Descargas).
  \item \Accion Ábrelo:
    \begin{itemize}
      \item \textbf{Windows:} doble clic en \texttt{Iniciar\_Tracker\_UAI.bat}.
      \item \textbf{macOS/Linux:} en la terminal, \texttt{bash iniciar\_tracker.sh}.
    \end{itemize}
\end{enumerate}

\begin{ojo}
  La \textbf{primera vez} tarda unos minutos (instala lo necesario) y luego abre
  en el navegador. Si Windows muestra \emph{«Windows protegió tu PC»}, pulsa
  \textbf{«Más información» \textrightarrow{} «Ejecutar de todas formas»}.
\end{ojo}

\section{Análisis paso a paso}
%%=============================================================================

\subsection*{Etapa 1 --- Video e intervalo}
\Accion Sube el video con \textbf{«Sube tu video»}. Con los botones
(\texttt{-10}/\texttt{-5}/\texttt{-1}/\texttt{+1}/\texttt{+5}/\texttt{+10}) o el
campo \textbf{«Ir al frame»}, ubica el inicio del movimiento y pulsa \textbf{«Usar
frame actual como INICIO»}; ve al final y pulsa \textbf{«…como FINAL»}.
\Verifica Aparece el mensaje verde con el intervalo elegido (arriba se ve esta
etapa en la imagen de la interfaz).

\subsection*{Etapa 2 --- Calibración}
\Accion Marca los dos extremos de tu regla (Punto 1 y Punto 2).
\Ingresa la \textbf{distancia real} entre ellos, en \textbf{metros}.
\Accion Marca el \textbf{Origen} y pulsa \textbf{«Aplicar calibración»}.
\Verifica Sale un mensaje verde con la escala (m/px).

\begin{center}
  \fcolorbox{uaiBorde}{white}{\includegraphics[width=0.92\linewidth]{ejemplos/ui_calibracion.png}}\\[2pt]
  {\footnotesize\color{uaiAcero}\faCamera\ Etapa 2: dos puntos de la escala, el
  origen y los ejes marcados; abajo, la calibración aplicada (m/px).}
\end{center}

\begin{ojo}
  Calibrar bien es lo más importante: si la distancia no está en metros o marcas
  mal la regla, \textbf{todos} los resultados saldrán mal.
\end{ojo}

\subsection*{Etapa 3 --- Seguimiento del objeto}
\Accion Modo \textbf{Automático}: elige \emph{Plantilla} (arrastra un recuadro
sobre el objeto) o \emph{Color} (clic sobre el objeto) y pulsa \textbf{«Ejecutar
seguimiento»}. \Verifica Revisa las detecciones superpuestas.
\Accion Corrige lo que esté mal: navega al frame y \textbf{haz clic en la
posición correcta}, o edita la tabla. (También existe el modo \textbf{Manual}:
clic sobre el objeto en cada frame.)

\begin{center}
  \includegraphics[width=\linewidth]{ejemplos/resultados/carro_overlay.png}\\[2pt]
  {\footnotesize\color{uaiAcero}\faCamera\ Ejemplo real: seguimiento del carro sobre el riel
  (trayectoria en cian; los puntos se separan a medida que acelera).}
\end{center}

\subsection*{Etapa 4 --- Cinemática}
\Verifica Aparecen los gráficos de \textbf{posición, velocidad, aceleración y
trayectoria}. Si se ven ruidosos, activa \textbf{«Suavizar posición»}.

\begin{center}
  \includegraphics[width=\linewidth]{ejemplos/resultados/carro_fit.png}\\[2pt]
  {\footnotesize\color{uaiAcero}\faCamera\ Ejemplo real: $x(t)$ parabólica ($R^2 = 0{,}99997$)
  y $v(t)$ lineal; ambas rutas dan la misma aceleración.}
\end{center}

\subsection*{Etapa 5 --- Ajuste del modelo}
\Accion Elige el \textbf{preset de tu laboratorio} (por ejemplo, plano inclinado
o péndulo). \Verifica El programa muestra el resultado (por ejemplo la
aceleración $a$) con su \textbf{incertidumbre} y el \textbf{$R^2$}. Anótalo.

\begin{center}
  \includegraphics[width=0.40\linewidth]{ejemplos/resultados/pendulo_overlay.png}\\[2pt]
  {\footnotesize\color{uaiAcero}\faCamera\ Ejemplo real (péndulo): órbita medida con la elipse
  ajustada, su centro ($+$) y focos ($\star$).}
\end{center}

\begin{idea}
  Para el gráfico final del grupo (por ejemplo $a$ vs $\sin\theta$), usa la
  herramienta \textbf{«Gráficos»}: elige la plantilla del laboratorio, escribe
  los pares de datos y obtén $g$ de la pendiente, con su incertidumbre.
\end{idea}

\subsection*{Etapa 6 --- Exportar}
\Accion Marca las figuras que quieres y el CSV, pulsa \textbf{«Generar archivos»}
y \textbf{«Descargar ZIP»}. \Guarda el ZIP (figuras + CSV) y una captura del
valor ajustado de cada lanzamiento; los necesitarás para el informe.

\begin{center}
  \fcolorbox{uaiBorde}{white}{\includegraphics[width=0.92\linewidth]{ejemplos/ui_exportar.png}}\\[2pt]
  {\footnotesize\color{uaiAcero}\faCamera\ Etapa 6: casillas de figuras a exportar y
  botón «Generar archivos».}
\end{center}

\section{Problemas frecuentes}
%%=============================================================================

\begin{table}[htbp]
  \centering
  \begin{tabular}{@{}p{5.2cm} p{8.3cm}@{}}
    \toprule
    \textbf{Problema} & \textbf{Solución} \\
    \midrule
    No abre & Reinstala Python con «Add to PATH». Si dice
      \texttt{ImportError}, corre otra vez \texttt{pip install -r requirements.txt}. \\
    El video no carga o se ve mal & Suele ser el códec del \texttt{.mov}.
      Conviértelo a mp4/H.264. Si es muy largo, recórtalo. \\
    No detecta bien el objeto & Prueba el método \emph{Color}, ajusta la
      tolerancia, o usa el modo \emph{Manual}. \\
    Puntos mal marcados & Navega al frame y haz clic en la posición correcta, o
      edita la tabla. \\
    Calibración rara & Revisa que la distancia esté en \textbf{metros}. \\
    Parece detenido & Espera: la precarga del video o la detección tardan un poco. \\
    No aparecen los gráficos & Necesitas calibrar y tener $\geq 2$ puntos en el
      intervalo. \\
    \bottomrule
  \end{tabular}
\end{table}

\section{Resumen: qué hacer, de principio a fin}
%%=============================================================================

\begin{resumen}
  \begin{enumerate}
    \item Abre el programa y sube el video (Etapa 1); elige el intervalo.
    \item Calibra con la regla (Etapa 2).
    \item Trackea y corrige los puntos (Etapa 3).
    \item Mira los gráficos (Etapa 4) y ajusta el modelo (Etapa 5); anota el
      resultado.
    \item Repite para cada lanzamiento; arma el gráfico del grupo en «Gráficos».
    \item Exporta y \textbf{guarda CSV + figuras + capturas} para el informe.
  \end{enumerate}
\end{resumen}
